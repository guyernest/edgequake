Some(self.cmp(other))

fn cmp(&self, other: &Self) -> Ordering {
other.score.partial_cmp(&self.score).unwrap_or(Ordering::Equal)

struct VectorStore {
documents: Vec<Document>,
}


impl VectorStore {
fn new() -> Self {
VectorStore {
documents: Vec::new(),
}
}


fn add_document(&amp;mut self, doc: Document) {
    self.documents.push(doc);
}

fn cosine_similarity(a: &amp;[f64], b: &amp;[f64]) -&gt; f64 {
    let dot: f64 = a.iter().zip(b.iter()).map(|(x, y)| x * y).sum();
    let mag_a: f64 = a.iter().map(|x| x * x).sum::&lt;f64&gt;().sqrt();
    let mag_b: f64 = b.iter().map(|x| x * x).sum::&lt;f64&gt;().sqrt();

    if mag_a == 0.0 || mag_b == 0.0 {
        return 0.0;
    }

    dot / (mag_a * mag_b)
}

fn search(&amp;self, query: &amp;[f64], top_k: usize) -&gt; Vec&lt;SearchResult&gt; {
    let mut results: Vec&lt;SearchResult&gt; = self
        .documents
        .iter()
        .map(|doc| SearchResult {
            id: doc.id.clone(),
            score: Self::cosine_similarity(query, &amp;doc.embedding),
        })
        .collect();

    results.sort_by(|a, b| {
        b.score
            .partial_cmp(&amp;a.score)
            .unwrap_or(Ordering::Equal)
    });
    results.truncate(top_k);
    results
}
}


}

Explanation

The cosine_similarity function computes three values: the dot product of the two vectors (sum of element-wise products), and the L2 norms (magnitudes) of each vector. Dividing the dot product by the product of the magnitudes yields the cosine of the angle between the vectors, which ranges from -1 (opposite) to 1 (identical direction). The zero-magnitude guard prevents division by zero for degenerate inputs.

The search function applies cosine similarity to every document, collects the results, sorts them in descending score order, and truncates to the requested top_k. This brute-force approach is O(n * d) where n is the number of documents and d is the embedding dimension -- simple and correct, but too slow for large corpora where approximate nearest neighbor indexes (like HNSW) are needed.
